🛠️ Bölüm 1: Pandas ETL — Kısa Örnek (Tarayıcıda Çalışır)

Bölüm 3'teki SuperMarket ETL'inin bellek içi versiyonu. Dosya okumak yerine StringIO kullanıyoruz; mantık aynı: temizle → birleştir → özetle.

import pandas as pd
from io import StringIO

orders_csv = """order_id,customer_id,amount,ts
1001,42,120.50,2024-06-01 09:14
1002,17,340.00,2024-06-01 10:02
1003,42,-50.00,2024-06-01 11:30
1004,,89.90,2024-06-01 12:11"""

customers_csv = """customer_id,name,city,email
17,Ali,Ankara,ali@ex.com
42,Ayşe,İzmir,ayse@ex.com
91,Veli,Ankara,veli@ex.com"""

# 1) EXTRACT
orders = pd.read_csv(StringIO(orders_csv))
customers = pd.read_csv(StringIO(customers_csv))
print(f"Ham: {len(orders)} sipariş, {len(customers)} müşteri")

# 2) TRANSFORM
orders = orders.dropna(subset=["customer_id", "amount"])
orders = orders[orders["amount"] > 0]
orders["ts"] = pd.to_datetime(orders["ts"])
customers = customers.drop(columns=["email"])  # KVKK: e-postayı düşür

df = orders.merge(customers, on="customer_id", how="inner")
report = (df.groupby("city")
            .agg(siparis=("order_id", "count"),
                 ort_sepet=("amount", "mean"),
                 ciro=("amount", "sum"))
            .reset_index()
            .sort_values("ciro", ascending=False))

# 3) LOAD (burada ekrana — gerçekte Parquet'e yazılır)
print("\n📊 Şehir raporu:")
print(report.to_string(index=False))

2b-2) Uzun örnek — sayfalı API + rate limit (Colab)

🎬 Senaryo: Open-Meteo hava kalitesi

Bir belediye, şehir bazında PM2.5 ortalamasını günlük raporlamak istiyor. API sayfalı değil ama birden fazla koordinat için döngü + time.sleep ile rate limit'e saygı gösteriyoruz — production ETL'de sık görülen pattern.

import requests
import pandas as pd
import time
from datetime import datetime, timezone

# Koordinatlar: Ankara, İstanbul, İzmir
cities = {
    "Ankara": (39.93, 32.85),
    "İstanbul": (41.01, 28.97),
    "İzmir": (38.42, 27.14),
}

rows = []
for city, (lat, lon) in cities.items():
    url = (
        "https://air-quality-api.open-meteo.com/v1/air-quality"
        f"?latitude={lat}&longitude={lon}"
        "&hourly=pm2_5&timezone=Europe%2FIstanbul&forecast_days=1"
    )
    r = requests.get(url, timeout=15)
    r.raise_for_status()
    data = r.json()
    pm = data["hourly"]["pm2_5"]
    rows.append({
        "city": city,
        "pm25_ort": sum(pm) / len(pm),
        "pm25_max": max(pm),
        "ornek_sayisi": len(pm),
        "extracted_at": datetime.now(timezone.utc).isoformat(),
    })
    time.sleep(0.5)  # API'yi yormamak için

report = pd.DataFrame(rows).sort_values("pm25_ort", ascending=False)
print(report.to_string(index=False))
report.to_parquet("warehouse/air_quality.parquet", index=False)

from pyspark.sql import functions as F

spark = SparkSession.builder.appName("city-etl").getOrCreate()

    (1001, 42, 120.50, "2024-06-01 09:14"),
    (1002, 17, 340.00, "2024-06-01 10:02"),
], ["order_id", "customer_id", "amount", "ts"])

🌊 Bölüm 4: Streaming Mantığı — Pandas ile Simülasyon (Kısa)

Kafka kurmadan Bölüm 4'teki 5 dakikalık pencere fikrini anlamak için: olayları zaman damgasına göre gruplayıp "son 5 dakikada kaç işlem?" sorusunu yanıtlayalım.

import pandas as pd

# Sentetik kart işlemleri (Bölüm 4 fraud sorusunun mini versiyonu)
events = pd.DataFrame({
    "card_id": ["C1","C1","C1","C2","C1","C1"],
    "city":    ["Ank","Ist","Ank","Ank","Ist","Izmir"],
    "amount":  [100, 200, 150, 50, 300, 80],
    "ts": pd.to_datetime([
        "2025-05-19 10:00", "2025-05-19 10:02", "2025-05-19 10:04",
        "2025-05-19 10:01", "2025-05-19 10:06", "2025-05-19 10:07"
    ])
})

# 5 dakikalık tumbling window — Spark'taki window() mantığı
events = events.sort_values("ts")
events["window_start"] = events["ts"].dt.floor("5min")

# Kural: aynı kart, aynı pencerede 3+ farklı şehir → şüpheli
suspicious = (events.groupby(["card_id", "window_start"])["city"]
              .nunique()
              .reset_index(name="sehir_sayisi"))
suspicious = suspicious[suspicious["sehir_sayisi"] >= 3]

print("Şüpheli kartlar (3+ şehir / 5 dk):")
print(suspicious if len(suspicious) else "  (bu mini veride yok — C1 için 10:00–10:05 penceresinde Ank+Ist=2 şehir)")

SparkSession.builder.config("spark.jars.packages", "org.apache.spark:spark-sql-kafka-0-10_2.12:3.5.0")

from pyspark.sql import SparkSession, functions as F
from pyspark.sql.types import StructType, StringType, DoubleType, TimestampType

spark = (SparkSession.builder
         .appName("fraud-stream")
         .config("spark.jars.packages",
                 "org.apache.spark:spark-sql-kafka-0-10_2.12:3.5.0")
         .getOrCreate())

schema = (StructType()
    .add("txn_id", StringType())
    .add("card_id", StringType())
    .add("city", StringType())
    .add("amount", DoubleType())
    .add("ts", TimestampType()))

raw = (spark.readStream
       .format("kafka")
       .option("kafka.bootstrap.servers", "localhost:9092")
       .option("subscribe", "card_txns")
       .load())

events = (raw
    .select(F.from_json(F.col("value").cast("string"), schema).alias("e"))
    .select("e.*"))

# 5 dk tumbling window + kart başına kaç farklı şehir?
by_window = (events
    .withWatermark("ts", "2 minutes")
    .groupBy(F.window("ts", "5 minutes"), "card_id")
    .agg(F.countDistinct("city").alias("sehir_sayisi"),
         F.collect_set("city").alias("sehirler")))

# Sadece şüpheli kartları filtrele
alarms = by_window.filter(F.col("sehir_sayisi") >= 3)

query = (alarms.writeStream
         .outputMode("update")
         .format("console")
         .option("truncate", False)
         .start())

query.awaitTermination()

📤 Producer — işlem olayları

from kafka import KafkaProducer
import json, time, random
from datetime import datetime, timezone

producer = KafkaProducer(
    bootstrap_servers="localhost:9092",
    value_serializer=lambda v: json.dumps(v).encode("utf-8"),
    key_serializer=lambda k: k.encode("utf-8"),
)

cities = ["Ankara", "İstanbul", "İzmir", "Bursa", "Antalya"]

# Kart C999: kasıtlı fraud — 5 dk içinde 3 şehir
fraud_pattern = [
    ("C999", "Ankara"), ("C999", "İstanbul"), ("C999", "İzmir"),
]
for i, (card, city) in enumerate(fraud_pattern + [("C100", "Ankara")] * 5):
    event = {
        "txn_id": 5000 + i,
        "card_id": card,
        "city": city,
        "amount": round(random.uniform(50, 800), 2),
        "ts": datetime.now(timezone.utc).isoformat(),
    }
    producer.send("card_txns", key=card, value=event)
    print("→", event)
    time.sleep(1)

producer.flush()

📥 Consumer — sliding window + alarm

from kafka import KafkaConsumer
from collections import defaultdict, deque
from datetime import datetime, timezone, timedelta
import json

WINDOW = timedelta(minutes=5)
consumer = KafkaConsumer(
    "card_txns",
    bootstrap_servers="localhost:9092",
    group_id="fraud-detector-v1",
    auto_offset_reset="earliest",
    value_deserializer=lambda v: json.loads(v.decode("utf-8")),
)

# card_id → deque of (timestamp, city)
history = defaultdict(deque)

def parse_ts(s):
    return datetime.fromisoformat(s.replace("Z", "+00:00"))

for msg in consumer:
    e = msg.value
    card = e["card_id"]
    ts = parse_ts(e["ts"])
    city = e["city"]

    q = history[card]
    q.append((ts, city))
    # Eski olayları pencereden at (watermark benzeri)
    while q and ts - q[0][0] > WINDOW:
        q.popleft()

    cities_in_window = {c for _, c in q}
    if len(cities_in_window) >= 3:
        print(f"🚨 ALARM | kart={card} | şehirler={cities_in_window} | son {WINDOW}")

6a) SHA-256 hash (geri döndürülemez)

import hashlib

def hash_email(email: str, salt: str = "bdt2025") -> str:
    raw = (salt + email.strip().lower()).encode("utf-8")
    return hashlib.sha256(raw).hexdigest()[:16]  # kısa gösterim

print(hash_email("ayse@ex.com"))
print(hash_email("ayse@ex.com"))  # aynı girdi → aynı hash
print(hash_email("veli@ex.com"))  # farklı girdi → farklı hash

6b) Maskeleme + k-anonymity (genelleştirme)

import pandas as pd

df = pd.DataFrame({
    "yas": [34, 36, 28, 55, 34],
    "sehir": ["Ankara", "Ankara", "İzmir", "Ankara", "Ankara"],
    "hastalik": ["grip", "grip", "grip", "diyabet", "grip"],
})

# k=2 için: yaş aralığına indirgeme
df["yas_grup"] = pd.cut(df["yas"], bins=[0, 30, 40, 100], labels=["≤30", "31-40", "41+"])
anon = df.drop(columns=["yas"])

# Her (yas_grup, sehir) en az 2 kişi mi?
counts = anon.groupby(["yas_grup", "sehir"]).size()
print("Grup büyüklükleri:\n", counts)
print("k-anonymity sağlandı mı?", (counts >= 2).all())

6c) Diferansiyel gizlilik — Laplace gürültüsü

import numpy as np

gercek_ortalama_maas = 42_000
epsilon = 1.0  # küçük ε → daha fazla gürültü, daha güçlü gizlilik
duyarlilik = 5_000  # tek kişi eklenince/çıkınca ortalamadaki max değişim (basitleştirilmiş)

gurultu = np.random.laplace(loc=0, scale=duyarlilik / epsilon)
yayimlanan = gercek_ortalama_maas + gurultu

print(f"Gerçek ortalama:     {gercek_ortalama_maas:,.0f} TL")
print(f"Eklenen gürültü:     {gurultu:+,.0f} TL")

🏪 Bölüm 7: Uzun Senaryo — E-Ticaret Veri Hattı (Uçtan Uca)

🎬 Senaryo: TrendShop

Her gece iki kaynaktan veri geliyor: orders (CSV) ve customers (JSON). Pazarlama "KVKK uyumlu şehir raporu" istiyor. Akış: temizle → anonimleştir → Parquet'e yaz → ertesi gün Spark/Pandas ile oku.

"""
TrendShop nightly ETL — Bölüm 3 + 4 + 5 birleşimi
Colab'da çalıştırın: !pip install pyarrow
"""
import pandas as pd
import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path

# ── EXTRACT ──
orders = pd.read_csv("data/orders.csv")
with open("data/customers.json") as f:
    customers = pd.DataFrame(json.load(f))

print(f"Extract: {len(orders)} sipariş, {len(customers)} müşteri")

# ── TRANSFORM ──
# 1) Veri kalitesi
orders = orders.dropna(subset=["customer_id", "amount"])
orders = orders[orders["amount"] > 0]
orders["ts"] = pd.to_datetime(orders["ts"])

# 2) KVKK — doğrudan kimlik ve e-posta düşür; customer_id hash'le
def pseudo_id(val, salt="trendshop"):
    return hashlib.sha256(f"{salt}:{val}".encode()).hexdigest()[:12]

customers = customers.drop(columns=["email", "name"], errors="ignore")
customers["customer_id"] = customers["customer_id"].apply(pseudo_id)
orders["customer_id"] = orders["customer_id"].apply(pseudo_id)

# 3) Join + aggregate
df = orders.merge(customers, on="customer_id", how="inner")
report = (df.groupby("city")
            .agg(siparis=("order_id", "count"),
                 ort_sepet=("amount", "mean"),
                 ciro=("amount", "sum"))
            .reset_index())

report["etl_run_at"] = datetime.now(timezone.utc).isoformat()
report["k_anon_not"] = "city düzeyinde; birey yok"

# ── LOAD ──
out = Path("warehouse/city_sales")
out.mkdir(parents=True, exist_ok=True)
pq_file = out / f"report_{datetime.now():%Y%m%d}.parquet"
report.to_parquet(pq_file, index=False)

print(f"Load: {pq_file} ({pq_file.stat().st_size} byte)")
print(report)

# ── Ertesi gün analiz (Bölüm 6'ya köprü) ──
loaded = pd.read_parquet(pq_file)
print("\nEn yüksek ciro:", loaded.loc[loaded["ciro"].idxmax(), "city"])

📡 Bölüm 8: Uzun Senaryo — Akıllı Şehir IoT Sensör Hattı

🎬 Senaryo: CitySense

Binlerce sıcaklık sensörü her dakika JSON satırı üretiyor. Batch ETL gece çalışıyor: ham log → temiz tablo → bölge ortalaması → Parquet. Bölüm 3 ETL + Bölüm 4 streaming arasındaki köprü: şimdilik batch, yarın Kafka'ya geçeriz.

import json
import pandas as pd
import numpy as np
from pathlib import Path
from datetime import datetime, timezone

np.random.seed(7)
# Ham sensör satırları (gerçekte Kafka / dosya akışından gelir)
raw_lines = []
for i in range(500):
    raw_lines.append(json.dumps({
        "sensor_id": f"S{i%50:03d}",
        "district": np.random.choice(["Cankaya", "Kecioren", "Yenimahalle"]),
        "temp_c": round(np.random.normal(22, 4), 2),
        "ts": pd.Timestamp("2025-05-19") + pd.Timedelta(minutes=i),
    }))

# EXTRACT
records = [json.loads(line) for line in raw_lines]
df = pd.DataFrame(records)

# TRANSFORM — aykırı sıcaklık, eksik sensör
df = df[(df["temp_c"] >= -10) & (df["temp_c"] <= 50)]
df["hour"] = df["ts"].dt.floor("h")

hourly = (df.groupby(["district", "hour"])
            .agg(okuma=("temp_c", "count"),
                 ort_sicaklik=("temp_c", "mean"),
                 max_sicaklik=("temp_c", "max"))
            .reset_index())

hourly["etl_run_at"] = datetime.now(timezone.utc).isoformat()

# LOAD
out = Path("warehouse/iot_hourly")
out.mkdir(parents=True, exist_ok=True)
hourly.to_parquet(out / "district_hourly.parquet", index=False)
print(hourly.groupby("district")["ort_sicaklik"].mean().sort_values(ascending=False))

📋 Bölüm 9: Web Sunucu Log Analizi — Kısa + Orta

Nginx/Apache logları metin dosyasıdır; ETL'in Extract adımı "satır oku", Transform "regex ile ayıkla" demektir. Bölüm 3'te CSV okuduk; format farklı, mantık aynı.

9a) Kısa — regex ile parse (tarayıcıda)

import re
import pandas as pd
from io import StringIO

# Tipik combined log format (IP gizlenmiş)
log = """192.168.1.1 - - [19/May/2025:10:00:01 +0300] "GET /api/orders HTTP/1.1" 200 1234
192.168.1.2 - - [19/May/2025:10:00:02 +0300] "POST /login HTTP/1.1" 401 89
192.168.1.1 - - [19/May/2025:10:00:05 +0300] "GET /api/orders HTTP/1.1" 200 1230
192.168.1.3 - - [19/May/2025:10:00:08 +0300] "GET /health HTTP/1.1" 200 12"""

pattern = re.compile(
    r'"(\w+) (\S+) HTTP.*?" (\d+) (\d+)'
)

rows = []
for line in log.strip().split("\n"):
    m = pattern.search(line)
    if m:
        rows.append({
            "method": m.group(1),
            "path": m.group(2),
            "status": int(m.group(3)),
            "bytes": int(m.group(4)),
        })

df = pd.DataFrame(rows)
print("Endpoint başına istek:")
print(df.groupby("path").size().sort_values(ascending=False))
print("\n4xx/5xx oranı:", (df["status"] >= 400).mean())

9b) Orta — büyük log dosyası chunk okuma (Colab)

import pandas as pd

chunks = []
for chunk in pd.read_csv(
    "access.log",
    sep=r"\s+",
    header=None,
    names=["ip", "dash1", "dash2", "ts", "tz", "request", "status", "bytes"],
    chunksize=100_000,
    engine="python",
    on_bad_lines="skip",
):
    chunk["status"] = pd.to_numeric(chunk["status"], errors="coerce")
    summary = chunk.groupby("status").size()
    chunks.append(summary)

total = pd.concat(chunks, axis=1).fillna(0).sum(axis=1)
print("HTTP status dağılımı:\n", total.sort_index())
